# Dependencies for running the test suite.
#
# Everything here must resolve on every interpreter in the support range,
# 3.9 through 3.14, because tox installs this file into each matrix env.
# Keep dev-only tooling in requirements_dev.txt -- tox, mypy and build have all
# dropped 3.9, so they cannot live in this file.

-r requirements.txt
-e .

# pytest 9 requires Python 3.10+, so 3.9 stays on the 8.4.x line.
pytest==8.4.2; python_version == "3.9"
pytest==9.1.1; python_version >= "3.10"

# 7.10.x is the only line shipping wheels for both cp39 and cp314. Outside it one
# end of the range loses the C tracer and silently falls back to the much slower
# pure-Python one ("Coverage.py ... without C extension").
coverage==7.10.7
